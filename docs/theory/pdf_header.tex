% =====================================================================
%  Blue theme for the SiamRAM theory course
% =====================================================================
\usepackage{xcolor}
\definecolor{ramblue}{HTML}{12477F}   % deep blue  (headings, title banner)
\definecolor{ramblueM}{HTML}{2E74B5}  % medium blue (accents, rules, subhead)
\definecolor{rambluL}{HTML}{D7E6F7}   % light blue (banner subtitle text)
\definecolor{ramcodebg}{HTML}{EEF4FB} % very light blue (code background)
\definecolor{ramrule}{HTML}{9DBEE2}   % soft blue (code left rule)

% ---- paragraph spacing / numbering ----
\setcounter{secnumdepth}{-1}
\usepackage{parskip}
\usepackage{float}

% ---- coloured, accented headings ----
\usepackage{titlesec}
\titleformat{\chapter}[hang]
  {\fontsize{26}{30}\selectfont\bfseries\color{ramblue}}{}{0pt}{}
  [\vspace{3pt}{\color{ramblueM}\titlerule[2pt]}]
\titlespacing*{\chapter}{0pt}{6pt}{14pt}
\titleformat{\section}
  {\Large\bfseries\color{ramblue}}{}{0pt}{}
  [\vspace{1pt}{\color{rambluL}\titlerule[0.8pt]}]
\titlespacing*{\section}{0pt}{1.7ex plus .3ex}{0.7ex}
\titleformat{\subsection}
  {\large\bfseries\color{ramblueM}}{}{0pt}{}
\titlespacing*{\subsection}{0pt}{1.1ex plus .3ex}{0.5ex}

% ---- blue running header / footer ----
\usepackage{fancyhdr}
\setlength{\headheight}{16pt}
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\headrule}{\color{ramblueM}\hrule height 1pt\vspace{1pt}}
\fancyhead[L]{\small\color{ramblue}\nouppercase{\leftmark}}
\fancyhead[R]{\small\color{ramblue}\textbf{\thepage}}
\fancyfoot[C]{\small\color{ramblueM}SiamRAM \textemdash{} A Mathematical \& Theoretical Course}
\fancypagestyle{plain}{%
  \fancyhf{}\renewcommand{\headrulewidth}{0pt}\renewcommand{\headrule}{}%
  \fancyfoot[C]{\small\color{ramblueM}\textbf{\thepage}}}

\makeatletter
% Fixed small monospace so wide ASCII diagrams (up to ~118 cols) fit the text width.
\renewcommand{\verbatim@font}{\ttfamily\fontsize{7pt}{8.4pt}\selectfont}
% Allow line breaks after underscores so long file_names in tables wrap instead of overflowing.
\let\origunderscore\_
\renewcommand{\_}{\origunderscore\allowbreak}
% Shrink + recolour highlighted code (yaml/bash) once fancyvrb is loaded by pandoc.
\AtBeginDocument{\@ifpackageloaded{fancyvrb}{\fvset{fontsize=\scriptsize}}{}}
\makeatother

% ---- code blocks on a light-blue panel with a blue left rule ----
\definecolor{shadecolor}{HTML}{EEF4FB}   % highlighted (Shaded) blocks -> light blue
\usepackage{mdframed}
\global\mdfdefinestyle{ramcode}{%
  backgroundcolor=ramcodebg, linecolor=ramrule,
  linewidth=3pt, leftline=true, rightline=false, topline=false, bottomline=false,
  innerleftmargin=8pt, innerrightmargin=6pt, innertopmargin=5pt, innerbottommargin=5pt,
  skipabove=6pt, skipbelow=6pt, leftmargin=0pt, rightmargin=0pt,
  splittopskip=6pt, splitbottomskip=6pt}
\surroundwithmdframed[style=ramcode]{verbatim}

% ---- slightly smaller, wrapping tables ----
\usepackage{etoolbox}
\AtBeginEnvironment{longtable}{\small}

% ---- a designed blue title page ----
\usepackage{tikz}
\makeatletter
\renewcommand{\maketitle}{%
  \begin{titlepage}%
    \thispagestyle{empty}%
    \hyphenpenalty=10000\exhyphenpenalty=10000\relax
    \begin{tikzpicture}[remember picture,overlay]
      % top blue banner + thin accent stripe under it
      \fill[ramblue]  (current page.north west) rectangle ([yshift=-0.42\paperheight]current page.north east);
      \fill[ramblueM] ([yshift=-0.42\paperheight]current page.north west) rectangle ([yshift=-0.435\paperheight]current page.north east);
      % title (white) inside the banner
      \node[anchor=north west, text=white, align=left, text width=0.80\paperwidth,
            font=\bfseries\fontsize{34}{40}\selectfont]
        at ([xshift=0.10\paperwidth, yshift=-0.15\paperheight]current page.north west)
        {\@title};
      % subtitle (light blue) inside the banner
      \node[anchor=north west, text=rambluL, align=left, text width=0.80\paperwidth,
            font=\LARGE]
        at ([xshift=0.10\paperwidth, yshift=-0.30\paperheight]current page.north west)
        {A Complete, Self-Contained Mathematical \& Theoretical Course};
      % author / date (blue) below the banner
      \node[anchor=north west, text=ramblue, align=left, text width=0.80\paperwidth,
            font=\large]
        at ([xshift=0.10\paperwidth, yshift=-0.56\paperheight]current page.north west)
        {\@author};
      \node[anchor=north west, text=ramblueM, align=left, font=\normalsize]
        at ([xshift=0.10\paperwidth, yshift=-0.605\paperheight]current page.north west)
        {\@date};
      % footer accent line + note
      \fill[ramblueM] ([yshift=0.085\paperheight]current page.south west) rectangle ([yshift=0.091\paperheight]current page.south east);
      \node[anchor=south west, text=ramblueM, font=\small]
        at ([xshift=0.10\paperwidth, yshift=0.098\paperheight]current page.south west)
        {Reverse-engineered from the implementation \textbullet{} distractor mode \& spike watcher out of scope};
    \end{tikzpicture}%
  \end{titlepage}%
}
\makeatother
